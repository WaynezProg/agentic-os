I'm using the `writing-plans` skill to create the implementation plan.

**Goal:** 在不改變現有 logs 流程的前提下，補齊 session usage 收斂（token、provider、model、cost）與摘要 API，提供 dashboard 先行的 quota 監控能力。

**Architecture:** 新增 `usage.py` 統一定義資料模型、parser 與 store；`supervisor` 只在 session 結束時觸發 extraction 並 upsert；`api/client/cli` 僅讀 `usage_records` 資料，維持分層不互相耦合。

**Tech Stack:** FastAPI、Typer、sqlite migration、pydantic、pytest。

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Protocol
import hashlib


@dataclass(frozen=True)
class UsageRecord:
    session_id: str
    harness_id: str
    provider: str
    model: str
    run_profile: str
    cwd: str
    started_at: str
    ended_at: str
    input_tokens: int = 0
    output_tokens: int = 0
    total_tokens: int = 0
    cost_usd: float = 0.0
    currency: str = "USD"
    source: str = "fallback"
    raw_evidence: str = "-"

    @property
    def token_total(self) -> int:
        return self.input_tokens + self.output_tokens


class UsageParser(Protocol):
    harness_id: str
    source: str
    def extract(self, session_id: str, harness_id: str, lines: list[dict[str, object]], fallback_provider: str, fallback_model: str) -> UsageRecord:
        ...


class OpenclawUsageParser:
    harness_id = "openclaw"
    source = "openclaw"

    def extract(self, session_id, harness_id, lines, fallback_provider, fallback_model):
        input_tokens = output_tokens = 0
        cost_usd = 0.0
        for row in lines:
            payload = row.get("payload", {})
            usage = payload.get("usage", {})
            input_tokens += int(usage.get("input_tokens", 0) or 0)
            output_tokens += int(usage.get("output_tokens", 0) or 0)
            cost_usd += float(payload.get("cost", {}).get("usd", 0.0) or 0.0)
        total_tokens = input_tokens + output_tokens
        evidence = hashlib.sha256(str(lines).encode()).hexdigest()[:16]
        return UsageRecord(
            session_id=session_id,
            harness_id=harness_id,
            provider=fallback_provider,
            model=fallback_model,
            run_profile="default",
            cwd="unknown",
            started_at="N/A",
            ended_at="N/A",
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            total_tokens=total_tokens,
            cost_usd=cost_usd,
            source=self.source,
            raw_evidence=evidence,
        )


class FallbackUsageParser:
    harness_id = "any"
    source = "fallback"

    def extract(self, session_id, harness_id, lines, fallback_provider, fallback_model):
        evidence = hashlib.sha256(str(lines).encode()).hexdigest()[:16]
        return UsageRecord(
            session_id=session_id,
            harness_id=harness_id,
            provider=fallback_provider,
            model=fallback_model,
            run_profile="default",
            cwd="unknown",
            started_at="N/A",
            ended_at="N/A",
            raw_evidence=evidence,
        )


_PARSERS: list[UsageParser] = [OpenclawUsageParser(), FallbackUsageParser()]


def read_usage_parser(harness_id: str) -> UsageParser:
    for parser in _PARSERS:
        if parser.harness_id == "any" or parser.harness_id == harness_id:
            return parser
    return FallbackUsageParser()
```

```python
class UsageStore:
    def __init__(self, conn: sqlite3.Connection):
        self._conn = conn

    def init(self) -> None:
        self._conn.execute(
            """
            CREATE TABLE IF NOT EXISTS usage_records (
                session_id TEXT PRIMARY KEY,
                harness_id TEXT NOT NULL,
                provider TEXT NOT NULL,
                model TEXT NOT NULL,
                run_profile TEXT NOT NULL,
                cwd TEXT NOT NULL,
                started_at TEXT NOT NULL,
                ended_at TEXT NOT NULL,
                input_tokens INTEGER NOT NULL DEFAULT 0,
                output_tokens INTEGER NOT NULL DEFAULT 0,
                total_tokens INTEGER NOT NULL DEFAULT 0,
                cost_usd REAL NOT NULL DEFAULT 0.0,
                currency TEXT NOT NULL DEFAULT 'USD',
                source TEXT NOT NULL DEFAULT 'fallback',
                raw_evidence TEXT NOT NULL DEFAULT '-',
                updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
            )
            """
        )

    def upsert(self, row: UsageRecord) -> None:
        self._conn.execute(
            """
            INSERT INTO usage_records(session_id, harness_id, provider, model, run_profile, cwd, started_at, ended_at,
                                      input_tokens, output_tokens, total_tokens, cost_usd, currency, source, raw_evidence)
            VALUES(:session_id, :harness_id, :provider, :model, :run_profile, :cwd, :started_at, :ended_at,
                   :input_tokens, :output_tokens, :total_tokens, :cost_usd, :currency, :source, :raw_evidence)
            ON CONFLICT(session_id) DO UPDATE SET
              harness_id=excluded.harness_id,
              total_tokens=excluded.total_tokens,
              updated_at=CURRENT_TIMESTAMP
            """,
            row.__dict__,
        )
```

```python
from agentic_os.usage import UsageStore, read_usage_parser


def _collect_usage_for_session(self, session_id: str) -> None:
    session = self.store.get_session(session_id)
    parser = read_usage_parser(session.agent_id)
    lines = self.store.read_logs(session_id)
    usage = parser.extract(
        session_id=session_id,
        harness_id=session.agent_id,
        lines=[row for _, row in lines],
        fallback_provider=session.resolved_provider or "",
        fallback_model=session.resolved_model or "",
    )
    usage_store = UsageStore(self.store.connection)
    usage_store.upsert(usage)
```

```python
@app.get("/usage/sessions/{session_id}")
def usage_session(session_id: str) -> dict[str, object]:
    record = usage_store.get(session_id)
    if record is None:
        return {"session_id": session_id, "provider": "N/A", "total_tokens": "N/A", "cost_usd": "N/A"}
    return record


@app.get("/usage/summary")
def usage_summary(
    from_: str | None = Query(default=None, alias="from"),
    to: str | None = Query(default=None, alias="to"),
    harness_id: str | None = None,
    provider: str | None = None,
) -> dict[str, object]:
    rows = usage_store.list_summary(from_ts=from_, to_ts=to, harness_id=harness_id, provider=provider)
    return {"count": len(rows), "rows": rows}


@app.get("/usage/quotas")
def usage_quotas(scope: str = Query(default="daily")) -> dict[str, object]:
    if scope == "daily":
        return {"scope": "daily", "rows": usage_store.list_daily_totals()}
    if scope == "session":
        return {"scope": "session", "rows": usage_store.list_session_totals()}
    raise HTTPException(status_code=400, detail={"supported": ["daily", "session"]})
```
